cmake_minimum_required(VERSION 3.16)
project(rvd_railbench VERSION 0.2.0 LANGUAGES CXX)

if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
include(CTest)
option(RVD_SIMULATION_ONLY "Disable physical hardware execution" ON)
option(RVD_WINDOWS_TIMER "Request 1 ms Windows timer resolution" ON)
set(ONNXRUNTIME_ROOT "" CACHE PATH "ONNX Runtime package root (include/, lib/)")
set(_ort_default OFF)
if(ONNXRUNTIME_ROOT)
  set(_ort_default ON)
endif()
option(RVD_ENABLE_ONNX "Enable optional ONNX Runtime backend" ${_ort_default})
if(NOT RVD_SIMULATION_ONLY)
  message(FATAL_ERROR "Hardware drivers and interlocks are not implemented; use RVD_SIMULATION_ONLY=ON")
endif()

add_library(rvd_build_options INTERFACE)
target_compile_features(rvd_build_options INTERFACE cxx_std_17)
set(CMAKE_CXX_EXTENSIONS OFF)
# Keep the original arithmetic order, including in downstream controller adapters.
if(MSVC)
  target_compile_options(rvd_build_options INTERFACE /fp:precise /W4 /utf-8)
else()
  target_compile_options(rvd_build_options INTERFACE -fno-fast-math -ffp-contract=off -Wall -Wextra -Wpedantic)
endif()

add_library(pfcr_core src/pfcr_controller.cpp src/policy_builtin.cpp src/policy_onnx.cpp)
add_library(rvd::pfcr ALIAS pfcr_core)
target_include_directories(pfcr_core PUBLIC include)
target_link_libraries(pfcr_core PUBLIC rvd_build_options)
if(RVD_ENABLE_ONNX)
  find_path(ONNXRUNTIME_INCLUDE onnxruntime_cxx_api.h PATHS "${ONNXRUNTIME_ROOT}/include" NO_DEFAULT_PATH)
  find_library(ONNXRUNTIME_LIB onnxruntime PATHS "${ONNXRUNTIME_ROOT}/lib" NO_DEFAULT_PATH)
  if(NOT ONNXRUNTIME_INCLUDE OR NOT ONNXRUNTIME_LIB)
    message(FATAL_ERROR "RVD_ENABLE_ONNX requires ONNXRUNTIME_ROOT with headers and library")
  endif()
  add_library(rvd_onnx UNKNOWN IMPORTED)
  set_target_properties(rvd_onnx PROPERTIES IMPORTED_LOCATION "${ONNXRUNTIME_LIB}"
    INTERFACE_INCLUDE_DIRECTORIES "${ONNXRUNTIME_INCLUDE}")
  target_compile_definitions(pfcr_core PUBLIC PFCR_WITH_ONNXRUNTIME)
  target_link_libraries(pfcr_core PUBLIC rvd_onnx)
endif()

add_library(rvd_runtime src/core/artifact.cpp src/control/pfcr_adapter.cpp src/runtime/runtime.cpp src/runtime/config.cpp src/runtime/options.cpp src/runtime/telemetry.cpp)
add_library(rvd::runtime ALIAS rvd_runtime)
target_link_libraries(rvd_runtime PUBLIC pfcr_core)
add_executable(rail_main apps/rail_main.cpp)
target_link_libraries(rail_main PRIVATE rvd_runtime)
if(WIN32 AND RVD_WINDOWS_TIMER)
  target_compile_definitions(rail_main PRIVATE RVD_WINDOWS_TIMER)
  target_link_libraries(rail_main PRIVATE winmm)
endif()

if(BUILD_TESTING)
  add_executable(golden_test tests/golden_test.cpp)
  target_link_libraries(golden_test PRIVATE pfcr_core)
  add_test(NAME golden_builtin COMMAND golden_test ${CMAKE_CURRENT_SOURCE_DIR})
  set_tests_properties(golden_builtin PROPERTIES LABELS "regression" TIMEOUT 120)
  if(RVD_ENABLE_ONNX)
    add_test(NAME golden_onnx COMMAND golden_test ${CMAKE_CURRENT_SOURCE_DIR} --onnx)
    set_tests_properties(golden_onnx PROPERTIES LABELS "regression;onnx" TIMEOUT 120)
  endif()
endif()
if(RVD_ENABLE_ONNX AND WIN32)
  file(GLOB ORT_DLLS "${ONNXRUNTIME_ROOT}/lib/*.dll")
  foreach(t rail_main golden_test)
    if(TARGET ${t})
      foreach(dll ${ORT_DLLS})
        add_custom_command(TARGET ${t} POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
          "${dll}" $<TARGET_FILE_DIR:${t}>)
      endforeach()
    endif()
  endforeach()
endif()

if(BUILD_TESTING)
  function(rvd_test name category source)
    add_executable(${name} ${source})
    target_link_libraries(${name} PRIVATE rvd_runtime)
    target_include_directories(${name} PRIVATE tests)
    add_test(NAME ${name} COMMAND ${name} ${CMAKE_CURRENT_SOURCE_DIR})
    set_tests_properties(${name} PROPERTIES LABELS "${category}" TIMEOUT 120)
  endfunction()
  rvd_test(controller_artifact_test unit tests/unit/controller_artifact_test.cpp)
  rvd_test(frames_config_test unit tests/unit/frames_config_test.cpp)
endif()
